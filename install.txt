#!/bin/bash
# Billard Stream — Linux installation
# curl -sSL https://raw.githubusercontent.com/CarbonDEA/billard-stream/master/install.txt | bash

set -e

echo "🎱 Billard Stream — Installation"
echo "================================="

# Spørg om klubnavn og API URL FØRST
read -p "Indtast klubnavn [fbk]: " CLUB_NAME < /dev/tty
CLUB_NAME=${CLUB_NAME:-fbk}

read -p "Indtast API URL [https://www.wahl-it.dk/billard-stream/api/command.php]: " API_URL < /dev/tty
API_URL=${API_URL:-https://www.wahl-it.dk/billard-stream/api/command.php}

# Installer dependencies
echo "📦 Installerer dependencies..."
sudo apt-get update -qq
sudo apt-get install -y -qq python3 python3-pip ffmpeg curl

# Download klient fra GitHub (SSL virker altid)
echo "📥 Downloader stream client..."
sudo curl -sSL "https://raw.githubusercontent.com/CarbonDEA/billard-stream/master/client/stream-client.py" -o /usr/local/bin/stream-client
sudo chmod +x /usr/local/bin/stream-client

# Installer Python pakke
echo "📦 Installerer Python pakker..."
sudo pip3 install requests --break-system-packages -q

# Opret config
echo "⚙️  Opretter konfiguration..."
sudo mkdir -p /etc/billard-stream
echo "{\"klub\": \"$CLUB_NAME\", \"api_url\": \"$API_URL\"}" | sudo tee /etc/billard-stream/config.json > /dev/null

# Opret systemd service
echo "🛠️  Opretter systemd service..."
sudo tee /etc/systemd/system/billard-stream.service > /dev/null <<SERVICE
[Unit]
Description=Billard Stream Client
After=network.target
Wants=network.target

[Service]
Type=simple
User=root
ExecStart=/usr/bin/python3 /usr/local/bin/stream-client
Restart=on-failure
RestartSec=10

[Install]
WantedBy=multi-user.target
SERVICE

# Start service
echo "▶️  Starter service..."
sudo systemctl daemon-reload
sudo systemctl enable billard-stream
sudo systemctl start billard-stream

echo ""
echo "✅ Klienten kører i baggrunden og starter automatisk ved genstart"
echo "   Klub: $CLUB_NAME"
echo "   API:  $API_URL"
echo "   Log:  sudo journalctl -u billard-stream -f"
