#!/bin/bash
# Billard Stream — Linux installation
# Serveres via: https://www.wahl-it.dk/billard-stream/install.txt
# Brug: curl -sSL https://www.wahl-it.dk/billard-stream/install.txt | bash

set -e
echo "🎱 Billard Stream — Installation"
echo "================================="
if [[ "$OSTYPE" != "linux-gnu"* ]]; then echo "❌ Kun Linux."; exit 1; fi
if ! command -v python3 &> /dev/null; then
  if command -v apt &> /dev/null; then sudo apt update && sudo apt install -y python3 python3-pip
  elif command -v dnf &> /dev/null; then sudo dnf install -y python3 python3-pip
  else echo "❌ Installér Python 3 manuelt"; exit 1; fi
fi
if ! command -v ffmpeg &> /dev/null; then
  if command -v apt &> /dev/null; then sudo apt install -y ffmpeg
  elif command -v dnf &> /dev/null; then sudo dnf install -y ffmpeg
  else echo "❌ Installér FFmpeg manuelt"; exit 1; fi
fi
sudo mkdir -p /etc/billard-stream /var/log/billard-stream
sudo curl -sSL "https://www.wahl-it.dk/billard-stream/stream-client.py" -o /usr/local/bin/stream-client
sudo chmod +x /usr/local/bin/stream-client
pip3 install websocket-client pystray Pillow --quiet
if [ ! -f /etc/billard-stream/config.json ]; then
  echo '{"server":"wss://www.wahl-it.dk/billard-stream/ws","klub":"","token":""}' | sudo tee /etc/billard-stream/config.json > /dev/null
fi
echo ""; echo "✅ Installation gennemført!"
echo "📝 Log ind på https://www.wahl-it.dk/billard-stream/ og konfigurér."
